cmake_minimum_required(VERSION 3.15)
cmake_policy(SET CMP0091 NEW)
project(realesrgan_bridge LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
if(NOT NCNN_SOURCE_DIR OR NOT GLSLANGVALIDATOR_EXECUTABLE)
 message(FATAL_ERROR "Run tools/build_native.ps1 to prepare pinned ncnn and shader compiler")
endif()
# GPU worker is single-session; avoid a separate OpenMP runtime dependency.
set(NCNN_OPENMP OFF CACHE BOOL "" FORCE)
include(layers.cmake)
add_subdirectory("${NCNN_SOURCE_DIR}" ncnn)
set(SHADERS)
foreach(shader realesrgan_preproc realesrgan_postproc realesrgan_preproc_tta realesrgan_postproc_tta)
 foreach(variant normal fp16s int8s)
  set(flags)
  set(name "${shader}")
  if(NOT variant STREQUAL "normal")
   set(name "${shader}_${variant}")
   list(APPEND flags -DNCNN_fp16_storage=1)
  endif()
  if(variant STREQUAL "int8s")
   list(APPEND flags -DNCNN_int8_storage=1)
  endif()
  set(output "${CMAKE_CURRENT_BINARY_DIR}/${name}.spv.hex.h")
  add_custom_command(OUTPUT "${output}" COMMAND "${GLSLANGVALIDATOR_EXECUTABLE}"
   ${flags} -V -s -x -o "${output}" "${CMAKE_CURRENT_SOURCE_DIR}/upstream/${shader}.comp"
   DEPENDS "upstream/${shader}.comp" VERBATIM)
  list(APPEND SHADERS "${output}")
 endforeach()
endforeach()
add_library(realesrgan_bridge SHARED bridge.cpp upstream/realesrgan.cpp ${SHADERS})
target_include_directories(realesrgan_bridge PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(realesrgan_bridge PRIVATE RS_BUILD _CRT_SECURE_NO_WARNINGS)
target_link_libraries(realesrgan_bridge PRIVATE ncnn)
add_executable(native_tests native_tests.cpp)
target_link_libraries(native_tests PRIVATE realesrgan_bridge)
enable_testing()
add_test(NAME abi COMMAND native_tests)
if(RS_MODEL_DIR)
 add_test(NAME gpu COMMAND native_tests "${RS_MODEL_DIR}")
endif()
